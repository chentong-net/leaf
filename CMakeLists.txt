cmake_minimum_required(VERSION 3.22.1)

project(Leaf VERSION 1.0.0 LANGUAGES NONE)

option(LEAF_METADATA_ONLY
        "Generate application metadata and skip platform configuration" OFF)

set(LEAF_APPLICATION_NAME "leaf-app" CACHE STRING
        "File-safe application identifier")
set(LEAF_APPLICATION_DISPLAY_NAME "Leaf App" CACHE STRING
        "User-visible application name")
set(LEAF_APPLICATION_ID "net.chentong.leaf.app" CACHE STRING
        "Default reverse-DNS application identifier")

if(LEAF_APPLICATION_NAME STREQUAL "")
    message(FATAL_ERROR "LEAF_APPLICATION_NAME must not be empty")
endif()

if(NOT LEAF_APPLICATION_NAME MATCHES "^[A-Za-z0-9._-]+$")
    message(FATAL_ERROR
            "LEAF_APPLICATION_NAME may only contain letters, digits, '.', '_', and '-'")
endif()

if(LEAF_APPLICATION_DISPLAY_NAME STREQUAL "")
    message(FATAL_ERROR "LEAF_APPLICATION_DISPLAY_NAME must not be empty")
endif()

if(NOT LEAF_APPLICATION_ID MATCHES "^[A-Za-z0-9.-]+$" OR
   NOT LEAF_APPLICATION_ID MATCHES "\\.")
    message(FATAL_ERROR
            "LEAF_APPLICATION_ID must be a reverse-DNS identifier")
endif()

function(leaf_escape_shell_value input output)
    set(value "${input}")
    if(value MATCHES "[\r\n]")
        message(FATAL_ERROR "Application metadata must not contain line breaks")
    endif()
    string(REPLACE "\\" "\\\\" value "${value}")
    string(REPLACE "\"" "\\\"" value "${value}")
    string(REPLACE "$" "\\$" value "${value}")
    string(REPLACE "`" "\\`" value "${value}")
    set("${output}" "${value}" PARENT_SCOPE)
endfunction()

leaf_escape_shell_value("${LEAF_APPLICATION_NAME}" LEAF_APPLICATION_NAME_ENV)
leaf_escape_shell_value(
        "${LEAF_APPLICATION_DISPLAY_NAME}"
        LEAF_APPLICATION_DISPLAY_NAME_ENV
)
leaf_escape_shell_value("${LEAF_APPLICATION_ID}" LEAF_APPLICATION_ID_ENV)
leaf_escape_shell_value("${PROJECT_VERSION}" LEAF_APPLICATION_VERSION_ENV)

set(LEAF_METADATA_CONTENT
"# Generated by the Leaf root CMake configuration.
export LEAF_APPLICATION_NAME=\"${LEAF_APPLICATION_NAME_ENV}\"
export LEAF_APPLICATION_DISPLAY_NAME=\"${LEAF_APPLICATION_DISPLAY_NAME_ENV}\"
export LEAF_APPLICATION_ID=\"${LEAF_APPLICATION_ID_ENV}\"
export LEAF_APPLICATION_VERSION=\"${LEAF_APPLICATION_VERSION_ENV}\"
")

file(GENERATE
        OUTPUT "${CMAKE_BINARY_DIR}/metadata.env"
        CONTENT "${LEAF_METADATA_CONTENT}"
)

function(leaf_apply_application_metadata target)
    if(NOT TARGET ${target})
        message(FATAL_ERROR
                "leaf_apply_application_metadata requires an existing target")
    endif()

    target_compile_definitions(${target} PRIVATE
            "LEAF_APPLICATION_NAME=\"${LEAF_APPLICATION_NAME}\""
            "LEAF_APPLICATION_DISPLAY_NAME=\"${LEAF_APPLICATION_DISPLAY_NAME}\""
            "LEAF_APPLICATION_ID=\"${LEAF_APPLICATION_ID}\""
    )
endfunction()

if(LEAF_METADATA_ONLY)
    message(STATUS "Generated ${CMAKE_BINARY_DIR}/metadata.env")
else()
    set(CMAKE_C_STANDARD 17)
    set(CMAKE_CXX_STANDARD 17)
    set(CMAKE_CXX_STANDARD_REQUIRED ON)
    set(CMAKE_EXPORT_COMPILE_COMMANDS ON)

    set(LEAF_TARGET "" CACHE STRING
            "Leaf target: macos, windows, web, android, ios, or ohos")

    if(LEAF_TARGET STREQUAL "")
        if(CMAKE_SYSTEM_NAME STREQUAL "iOS")
            set(LEAF_TARGET "ios")
        elseif(EMSCRIPTEN)
            set(LEAF_TARGET "web")
        elseif(WIN32)
            set(LEAF_TARGET "windows")
        elseif(APPLE)
            set(LEAF_TARGET "macos")
        elseif(ANDROID)
            set(LEAF_TARGET "android")
        else()
            message(FATAL_ERROR
                    "Unable to select a default Leaf target. "
                    "Configure with -DLEAF_TARGET=<target>.")
        endif()
    endif()

    message(STATUS "Leaf target: ${LEAF_TARGET}")
    message(STATUS "Leaf application: ${LEAF_APPLICATION_NAME}")

    if(LEAF_TARGET STREQUAL "macos")
        add_subdirectory(macos)
    elseif(LEAF_TARGET STREQUAL "windows")
        add_subdirectory(windows)
    elseif(LEAF_TARGET STREQUAL "web")
        add_subdirectory(web)
    elseif(LEAF_TARGET STREQUAL "android")
        add_subdirectory(android/leaf-android/src/main/cpp)
    elseif(LEAF_TARGET STREQUAL "ios")
        add_subdirectory(ios/cmake)
    elseif(LEAF_TARGET STREQUAL "ohos")
        add_subdirectory(ohos/leaf_ohos/src/main/cpp)
    else()
        message(FATAL_ERROR
                "Unsupported Leaf target '${LEAF_TARGET}'. "
                "Expected macos, windows, web, android, ios, or ohos.")
    endif()
endif()
